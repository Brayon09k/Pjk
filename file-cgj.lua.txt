-- Controle do giro (velocidade: 15 graus por frame)
local spinConnection = nil
local function startSpin()
    if spinConnection then spinConnection:Disconnect() end
    spinConnection = RunService.RenderStepped:Connect(function()
        if cfg.Spin and LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("HumanoidRootPart") then
            local hrp = LocalPlayer.Character.HumanoidRootPart
            hrp.CFrame = hrp.CFrame * CFrame.Angles(0, math.rad(15), 0)
        end
    end)
end
local function stopSpin()
    if spinConnection then spinConnection:Disconnect() spinConnection = nil end
end

local function getRealColor()
    local r = math.max(0, BASE_R - cfg.R)
    local g = math.max(0, BASE_G - cfg.G)
    local b = math.max(0, BASE_B - cfg.B)
    return Color3.fromRGB(r, g, b)
end

-- --- BUSCA DE PARTES ---
local function getPrecisePart(char, partName)
    if partName == "Head" then return char:FindFirstChild("Head") 
    elseif partName == "Torso" then return char:FindFirstChild("UpperTorso") or char:FindFirstChild("Torso")
    elseif partName == "RightArm" then return char:FindFirstChild("RightUpperArm") or char:FindFirstChild("Right Arm")
    elseif partName == "LeftArm" then return char:FindFirstChild("LeftUpperArm") or char:FindFirstChild("Left Arm")
    elseif partName == "Legs" then return char:FindFirstChild("RightUpperLeg") or char:FindFirstChild("Right Leg") end
    return nil
end

-- --- LINHAS ---
local function createTracer()
    local line = Drawing.new("Line")
    line.Thickness = 1.5
    line.Transparency = 1
    line.Visible = false
    return line
end

local tracers = {}
local highlights = {}

local function clearESP(player)
    local line = tracers[player]
    if line then
        line.Visible = false
    end
    local hl = highlights[player]
    if hl then
        pcall(function() hl:Destroy() end)
        highlights[player] = nil
    end
end

local function updateESP(player, color)
    local char = player.Character
    local isTeammate = (player.Team == LocalPlayer.Team)
    local line = tracers[player]
    if not line then
        line = createTracer()
        tracers[player] = line
    end
    
    if not cfg.ESP then
        line.Visible = false
        return
    end
    
    if char and char:FindFirstChild("HumanoidRootPart") then
        local humanoid = char:FindFirstChild("Humanoid")
        local isAlive = humanoid and humanoid.Health > 0
        if not isAlive then
            clearESP(player)
            return
        end
        
        local root = char.HumanoidRootPart
        local pos, vis = Camera:WorldToViewportPoint(root.Position)
        local hl = highlights[player]
        
        if not (cfg.SafeTeam and isTeammate) then
            if not hl then
                hl = Instance.new("Highlight")
                hl.Name = "MzHighlight"
                hl.Parent = char
                highlights[player] = hl
            end
            hl.FillColor = isTeammate and Color3.new(0,1,0) or color
            hl.Enabled = true
            if vis then
                line.From = Vector2.new(Camera.ViewportSize.X/2, Camera.ViewportSize.Y)
                line.To = Vector2.new(pos.X, pos.Y)
                line.Color = isTeammate and Color3.new(0,1,0) or color
                line.Visible = true
            else
                line.Visible = false
            end
        else
            if hl then
                hl:Destroy()
                highlights[player] = nil
            end
            line.Visible = false
        end
    else
        clearESP(player)
    end
end

-- --- ARRASTAR ---
local function makeDraggable(topbar, object)
    local dragging, dragInput, dragStart, startPos
    topbar.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            dragging = true
            dragStart = input.Position
            startPos = object.Position
            input.Changed:Connect(function()
                if input.UserInputState == Enum.UserInputState.End then
                    dragging = false
                end
            end)
        end
    end)
    topbar.InputChanged:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch then
            dragInput = input
        end
    end)
    RunService.RenderStepped:Connect(function()
        if dragging and dragInput then
            local delta = dragInput.Position - dragStart
            object.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + delta.X, startPos.Y.Scale, startPos.Y.Offset + delta.Y)
        end
    end)
end

    if p == cAju then b.LayoutOrder = orderAju; orderAju = orderAju + 1 end

            cfg.Spin = not cfg.Spin

            cfg.AntiKick = not cfg.AntiKick

    end)
    RunService.RenderStepped:Connect(function()
        local act = (id == "AimbotForte" and cfg.AimbotMode == "Forte") or (id == "AimbotFraco" and cfg.AimbotMode == "Fraco") or (cfg[id] == true)

    end)

    local debounce = false
    local function updateValue(delta)
        if debounce then return end
        debounce = true
        local newVal = math.clamp(cfg[key] + delta, min, max)
        if cfg[key] ~= newVal then
            cfg[key] = newVal
            v.Text = isDecimal and string.format("%.2f", cfg[key]) or tostring(math.floor(cfg[key]))
        end
        task.wait(0.1)
        debounce = false
    end
    
    m.MouseButton1Down:Connect(function() updateValue(-(step or 5)) end)
    pl.MouseButton1Down:Connect(function() updateValue(step or 5) end)
end

-- LOG de kicks
local kickLogs = {}
local function logKick(msg)
    table.insert(kickLogs, 1, "• " .. tostring(msg))
    if #kickLogs > 5 then table.remove(kickLogs) end
    local txt = ""
    for _, l in pairs(kickLogs) do txt = txt .. l .. "\n" end
    if kickLogBox then kickLogBox.Text = txt end
end

-- AUTO REJOIN (volta para o mesmo servidor)
local function queueRejoin()
    if cfg._rejoining or not cfg.AutoReconnect then return end
    cfg._rejoining = true
    task.delay(4, function()
        pcall(function()
            TeleportService:TeleportToPlaceInstance(game.PlaceId, game.JobId, LocalPlayer)
        end)
        task.wait(8)
        cfg._rejoining = false
    end)
end

-- HOOK ANTI KICK
function setupAntiKick()
    if cfg._antiKickHooked then return true end

    local ok = pcall(function()
        local mt = getrawmetatable(game)
        if not mt then error("sem metatable") end
        local oldNamecall = mt.__namecall
        setreadonly(mt, false)
        mt.__namecall = newcclosure(function(self, ...)
            local method = getnamecallmethod()
            if method == "Kick" and self == LocalPlayer then
                if cfg.AntiKick then
                    local args = { ... }
                    logKick(tostring(args[1] or "sem mensagem"))
                    queueRejoin()
                    return
                end
            end
            return oldNamecall(self, ...)
        end)
        setreadonly(mt, true)
    end)

    if not ok then
        ok = pcall(function()
            local oldKick = hookfunction(LocalPlayer.Kick, function(self, msg)
                if cfg.AntiKick then
                    logKick(tostring(msg or "sem mensagem"))
                    queueRejoin()
                    return
                end
                return oldKick(self, msg)
            end)
        end)
    end

    cfg._antiKickHooked = ok
    if kickStatus then
        kickStatus.Text = ok and "Hook ativo" or "Hook indisponível (usar rejoin)"
        kickStatus.TextColor3 = ok and Color3.new(0,1,0) or Color3.new(1,0.6,0)
    end
    return ok
end

-- ========== LOOP PRINCIPAL (BLINDADO) ==========
RunService.RenderStepped:Connect(safeLoop(function()
    fovFrame.Visible = cfg.FOV
    fovFrame.Size = UDim2.new(0, cfg.FOVSize*2, 0, cfg.FOVSize*2)
    fovFrame.Position = UDim2.new(0.5, -cfg.FOVSize, 0.5, -cfg.FOVSize + cfg.FOVPos)

    local bgColor = getRealColor()
    local darkBg = Color3.new(bgColor.R * 0.15, bgColor.G * 0.15, bgColor.B * 0.15)
    main.BackgroundColor3 = darkBg
    confirmFrame.BackgroundColor3 = darkBg
    main.BackgroundTransparency = cfg.Transparency
    confirmFrame.BackgroundTransparency = cfg.Transparency

    local borderColor
    if cfg.LEDActive then
        local hue = (tick() * (cfg.LEDSpeed / 10)) % 1
        borderColor = Color3.fromHSV(hue, 1, 1)
    else
        borderColor = bgColor
    end
    
    fovStroke.Color = borderColor
    mStroke.Color = borderColor
    bStroke.Color = borderColor
    cStroke.Color = borderColor
    title.TextColor3 = borderColor

    if cfg.ESP then
        for _, player in pairs(Players:GetPlayers()) do
            if player ~= LocalPlayer then
                updateESP(player, borderColor)
            end
        end
    else
        for _, player in pairs(Players:GetPlayers()) do
            if player ~= LocalPlayer then
                clearESP(player)
            end
        end
    end

    local target = nil
    local dist = cfg.FOVSize
    local center = Vector2.new(Camera.ViewportSize.X/2, (Camera.ViewportSize.Y/2) + cfg.FOVPos)
    
    if cfg.AimbotMode ~= "None" and cfg.FOV then
        for _, player in pairs(Players:GetPlayers()) do
            if player ~= LocalPlayer then
                local char = player.Character
                local isTeammate = (player.Team == LocalPlayer.Team)
                if char and not (cfg.SafeTeam and isTeammate) then
                    local humanoid = char:FindFirstChild("Humanoid")
                    if humanoid and humanoid.Health > 0 then
                        local part = getPrecisePart(char, cfg.TargetPart)
                        if part then
                            local pos, vis = Camera:WorldToViewportPoint(part.Position)
                            if vis then
                                local mDist = (Vector2.new(pos.X, pos.Y) - center).Magnitude
                                if mDist <= cfg.FOVSize and mDist < dist then
                                    dist = mDist
                                    target = part
                                end
                            end
                        end
                    end
                end
            end
        end
    end
    
    if target then
        Camera.CFrame = Camera.CFrame:Lerp(CFrame.new(Camera.CFrame.Position, target.Position), (cfg.AimbotMode == "Forte" and 1 or 0.15))
    end
end))
